// fix-search-member-scope: 팀장(null)이면 빈 글. ?n · ?n+1 은 JSON 배열(json_each 로 풂)
function searchScopeSql(scope, n){
	return scope ? ` AND ("cc" IN (SELECT value FROM json_each(?${n})) OR "bcc" IN (SELECT value FROM json_each(?${n + 1})))` : ''
}

function searchScopeParams(scope){
	return scope ? [JSON.stringify(scope.ccs), JSON.stringify(scope.bccs)] : []
}

// fix-search-member-scope: 범위가 있을 때만 바인딩 (팀장의 문장은 그대로)
function searchScopeBind(stmt, scope){
	return scope ? stmt.bind(...searchScopeParams(scope)) : stmt
}
